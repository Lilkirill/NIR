# Архитектура проекта

# 

# 

# app/api              — программный интерфейс HTTP

# app/core             — настройки, база данных, модели и схемы данных

# app/parsers          — парсеры конфигураций

# app/vendors          — вендорные адаптеры

# app/validators       — проверка схемы, правил и тестов

# app/services         — бизнес-логика приложения

# app/reports          — генерация отчётов

# rules                — правила валидации в формате YAML

# schemas              — JSON-схема нормализованной конфигурации

# templates            — HTML-шаблоны Jinja2

# configs              — примеры и датасеты конфигураций

# tests                — автоматизированные тесты

# scripts              — обучение и проверка датасета для Qwen

# tools                — импорт внешних датасетов

# grafana              — панели мониторинга и автоматическая настройка Grafana
